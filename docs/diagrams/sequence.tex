% ---------- 1. Start   (lanes: 0 op, 1 sb, 2 ue, 3 k8s, 4 slice, 5 ptp, 6 nep, 7 capif, 8 gw, 9 planes, 10 llm)

\msg{8}{9}{27.6}{route to plane}
